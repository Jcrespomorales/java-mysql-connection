-- 1. Crear la base de datos
CREATE DATABASE IF NOT EXISTS concesionario;
USE concesionario;

-- 2.a. Crear tabla empleados
CREATE TABLE empleados (
    nif VARCHAR(9) PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    apellido1 VARCHAR(50) NOT NULL,
    apellido2 VARCHAR(50),
    direccion VARCHAR(100),
    telefono VARCHAR(15),
    email VARCHAR(100),
    sueldo DECIMAL(10,2)
);

-- 2.b. Crear tabla clientes
CREATE TABLE clientes (
    nif VARCHAR(9) PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    apellido1 VARCHAR(50) NOT NULL,
    apellido2 VARCHAR(50),
    direccion VARCHAR(100),
    telefono VARCHAR(15),
    email VARCHAR(100)
);

-- 2.c. Crear tabla coches
CREATE TABLE coches (
    matricula VARCHAR(10) PRIMARY KEY,
    modelo VARCHAR(50) NOT NULL,
    tipo_vehiculo ENUM('km0', 'segunda_mano', 'nuevo') NOT NULL,
    precio DECIMAL(10,2),
    color VARCHAR(30),
    kilometros INT,
    garantia_meses INT, -- Puede quedar NULL si no tiene
    antiguo_propietario VARCHAR(100) -- Puede quedar NULL si es nuevo o km0
);
